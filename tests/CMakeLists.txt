cmake_minimum_required(VERSION 3.24)
project(InfiniPaintPenTests LANGUAGES C CXX)
enable_testing()
add_executable(brush_pressure brush_pressure.cpp)
target_include_directories(brush_pressure PRIVATE ../include)
target_compile_features(brush_pressure PRIVATE cxx_std_20)
add_test(NAME brush_pressure COMMAND brush_pressure)
add_executable(pen_pressure pen_pressure.cpp)
target_compile_features(pen_pressure PRIVATE cxx_std_20)
add_test(NAME pen_pressure COMMAND pen_pressure)
add_executable(pen_filter pen_stabilizer.cpp)
target_compile_features(pen_filter PRIVATE cxx_std_20)
if(PENTRACE_REFERENCE)
    target_sources(pen_filter PRIVATE "${PENTRACE_REFERENCE}/src/core.cpp")
    target_include_directories(pen_filter PRIVATE "${PENTRACE_REFERENCE}/src")
    target_compile_definitions(pen_filter PRIVATE PENTRACE_REFERENCE=1)
endif()
add_test(NAME pen_filter COMMAND pen_filter)

if(WIN32 AND SDL_SOURCE)
    set(SDL_SHARED ON CACHE BOOL "" FORCE)
    set(SDL_STATIC OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    add_subdirectory("${SDL_SOURCE}" sdl)
    add_executable(pen_history pen_history.c)
    target_link_libraries(pen_history PRIVATE SDL3::SDL3)
    add_custom_command(TARGET pen_history POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:pen_history>)
    add_test(NAME pen_history COMMAND pen_history)
endif()
