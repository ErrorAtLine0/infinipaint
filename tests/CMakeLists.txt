cmake_minimum_required(VERSION 3.24)
project(InfiniPaintPenTests LANGUAGES C CXX)
enable_testing()
if(WIN32 AND SDL_SOURCE)
    set(SDL_SHARED ON CACHE BOOL "" FORCE)
    set(SDL_STATIC OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    add_subdirectory("${SDL_SOURCE}" sdl)
    add_executable(pen_history pen_history.c)
    target_link_libraries(pen_history PRIVATE SDL3::SDL3)
    add_custom_command(TARGET pen_history POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:pen_history>)
    add_test(NAME pen_history COMMAND pen_history)
endif()
